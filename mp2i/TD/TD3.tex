\begin{Exercise}[title={Pointeurs}]\\
	On considère le programme suivant :
	\begin{langageC*}{fontsize=\small, tabsize=0}
		int a = 4;
		int b = 1;
		int c = 2;
		int* p;
		int* q;
		p = &a;
		q = &c;
		*p = *q + 1;
		p = q;
		q = &b;
		*p = *p - *q;
		*q = *q + 1;
		*p *= *q;

\begin{Exercise}[title={Fuite mémoire (\textit{Memory leak})}]\\
	On considère le code C suivant :
	\begin{langageC*}{fontsize=\small, tabsize=0}
		#include <stdlib.h>
		
		void simulations() {
		    int *donnees;
		    for (int i = 0; i < 1000; i++) {
		        donnees = malloc(10000 * sizeof(int));
		        // Traitement sur le tableau donnees (non détaillé ici)
		    }
		    free(donnees);
		}
	\end{langageC*}
	\Question{Ce programme compile sans erreur. Pourtant, il pose un grave problème de gestion de la mémoire. Lequel ?}
	\Question{Combien de blocs mémoires ont été alloués ? Combien ont été correctement libérés ?}
	\Question{Proposer une correction de la fonction {\tt simulations} pour corriger cette erreur.}
\end{Exercise}
