\begin{Exercise}[title={Arborescence}]\\
On donne ci-dessous l'arborescence du répertoire personnel de l'utilisateur {\tt John} : 
\begin{center}
	\pstree[levelsep=1.5cm]{\noeud{John}}{
						\pstree[levelsep=1.2cm]{\noeud{Public}}{\Tr{\tt img1.jpg} \Tr{\tt img2.jpg} \Tr{\tt img3.jpg}}
						\noeud{Téléchargements}
						\pstree[levelsep=1.2cm]{\noeud{Documents}}{\Tr{\tt secret.txt} \Tr{\tt .abc}}
						\noeud{Vidéos}
						\pstree[levelsep=1.2cm]{\noeud{Travail}}{\noeud{\footnotesize Cours} \noeud{\footnotesize TD} \noeud{\footnotesize TP}}}
\end{center}

\Question{On suppose maintenant qu'on se trouve dans le répertoire {\tt Documents}. Donner l'effet des commandes suivantes (exécutées dans l'ordre) :}
\subQuestion{\tt touch new.txt}
\subQuestion{\tt ls }
\subQuestion{\tt mkdir Todo}
\subQuestion{\tt cd Todo}
\subQuestion{\tt cp \~{}/Public/img1.jpg .}
\subQuestion{\tt ln \~{}/Public/img2.jpg img1-ln.jpg}
\subQuestion{\tt ln -s \~{}/Public/img3.jpg img3-ls.jpg}
\Question{En supposant qu'on se trouve dans le dossier {\tt Cours}, donner les chemins relatifs pour accéder aux dossiers suivants :}
\subQuestion{\tt Travail}
\subQuestion{\tt Documents}

\end{Exercise}

\begin{Exercise}[title={Tubes et composition de commandes}]\\
Sous Linux, le symbole {\tt |} (\textit{pipe}) permet de connecter la sortie standard d'une commande à l'entrée standard de la suivante.

\Question{On rappelle les commandes suivantes :}
\begin{itemize}
	\item {\tt wc -l} : compte le nombre de lignes reçues en entrée ;
	\item {\tt grep motif} : filtre et ne conserve que les lignes contenant le motif ;
	\item {\tt sort} : trie les lignes reçues par ordre alphabétique ;
	\item {\tt head -n k} : affiche les $k$ premières lignes reçues en entrée.
\end{itemize}

\Question{Expliquer ce que réalise chacune des commandes composées ci-dessous :}
\subQuestion{\tt ls -l /usr/bin | wc -l}
\subQuestion{\tt cat eleves.txt | grep "MP2I" | sort}
\subQuestion{\tt ps aux | grep "firefox" | wc -l}
\subQuestion{\tt ls -S /var/log | head -n 5}

\Question{Écrire une ligne de commande permettant de compter le nombre de fichiers portant l'extension {\tt .c} présents dans le répertoire courant.}
\end{Exercise}

\begin{Exercise}[title={Inodes et suppression de liens}]\\
Dans un terminal, un utilisateur exécute la suite d'instructions suivante :
\begin{minted}{shell}
$ echo "Données importantes" > original.txt
$ ln original.txt hardlink.txt
$ ln -s original.txt symlink.txt
\end{minted}

\Question{Rappeler la différence fondamentale de structure entre {\tt hardlink.txt} et {\tt symlink.txt} vis-à-vis du nœud d'index (\textit{inode}) du fichier initial.}
\Question{On exécute maintenant la commande {\tt rm original.txt}.}
\subQuestion{Quel est l'effet de la commande {\tt cat hardlink.txt} ? Expliquer pourquoi.}
\subQuestion{Quel est l'effet de la commande {\tt cat symlink.txt} ? Comment qualifie-t-on alors ce lien symbolique ?}
\subQuestion{Quand les blocs de données initialement alloués sur le disque seront-ils réellement libérés par le système de fichiers ?}
\end{Exercise}
