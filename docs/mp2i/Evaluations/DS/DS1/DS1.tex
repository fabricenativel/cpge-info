\begin{Exercise}[title = {Questions de cours}] \\
	On considère l'algorithme suivant : \\
	\begin{algorithm}[H]
		\DontPrintSemicolon
		\caption{Multiplier sans utiliser {\tt *}}
		\Entree{$n \in \N, m \in \N$}
		\Sortie{$nm$}
		\everypar={\footnotesize \textcolor{gray}{\nl}}
		$r \leftarrow 0$\;
		\Tq{$m > 0$}{
			$m \leftarrow m-1$ \;
			$r \leftarrow r+n$ \;
		}
		\Return $r$
	\end{algorithm}
	\Question{Donner les valeurs successives prises par les variables $m$, $n$ et $r$ si l'on exécute cet algorithme avec $n=7$ et $m=4$. On pourra recopier et compléter le tableau suivant :\\
		\begin{tabular}{|l|c|c|c|}
			\cline{2-4}
			\multicolumn{1}{l|}{}        & $n$      & $m$      & $r$       \\
			\hline
			valeurs initiales            & 7        & 4        & 0         \\
			\hline
			après un tour de boucle      & \comp{7} & \comp{3} & \comp{7}  \\
			\hline
			après deux tours de boucle   & \comp{7} & \comp{2} & \comp{14} \\
			\hline
			après trois tours de boucle  & \comp{7} & \comp{1} & \comp{21} \\
			\hline
			après quatre tours de boucle & \comp{7} & \comp{0} & \comp{28} \\
			\hline
		\end{tabular}
	}
	\Question{Donner une implémentation de cet algorithme en langage C sous la forme d'une fonction {\tt multiplie} de signature \mintinline{c}{int multiplie(int n, int m)}. On précisera soigneusement la spécification de cette fonction en commentaire dans le code et on vérifiera les préconditions à l'aide d'instructions {\tt assert}.}
	\ifcorrige
		\corpartC{mult.c}{}{}{4}{15}
	\fi
	\Question{Donner la définition d'un variant de boucle, puis prouver que cet algorithme termine.}
	\tcor{Un \textit{variant de boucle} est une quantité qui dépend des variables du programme et qui est :
		\begin{enumerate}
			\item entière,
			\item positive,
			\item strictement décroissante.
		\end{enumerate}
		Dans l'algorithme ci-dessus, la quantité $m$ est un variant de boucle ; en effet :
		\begin{enumerate}
			\item $m \in \N$ (par précondition).
			\item $m$ reste positif ou nul : par condition d'entrée dans la boucle $m \geqslant 1$, et comme on décrémente $m$ de 1, $m$ reste $\geqslant 0$ après le tour de boucle.
			\item $m$ décroît strictement car il est diminué de 1 lors de chaque passage dans la boucle.
		\end{enumerate}
		L'algorithme termine car on a trouvé un variant de boucle.
	}
	\Question{Donner la définition d'un invariant de boucle, puis prouver que cet algorithme est correct.}
	\tcor{Un \textit{invariant de boucle} est une propriété qui dépend des variables du programme et qui :
		\begin{enumerate}
			\item est vraie avant d'entrer dans la boucle (initialisation),
			\item reste vraie après un tour de boucle si elle l'était au début du tour (conservation).
		\end{enumerate}
		En sortie de boucle, la validité d'un invariant permet de prouver la correction de l'algorithme.\\
		En notant $m_0$ la valeur initiale de $m$, montrons que la propriété $I$ : \og $r = (m_0 - m)n$\fg{} est un invariant de boucle.
		\begin{enumerate}
			\item Avant d'entrer dans la boucle, $m = m_0$ donc $(m_0 - m)n = 0$. Comme $r$ est initialisé à $0$, la propriété $I$ est vérifiée.
			\item On suppose $I$ vérifiée au début d'un tour de boucle et on note $r'$ (resp. $m'$) les valeurs prises par $r$ (resp. $m$) à la fin de ce tour. On a $m' = m - 1$ et $r' = r + n$.
			      Calculons : $(m_0 - m')n = (m_0 - (m - 1))n = (m_0 - m + 1)n = (m_0 - m)n + n$. 
			      Or, par hypothèse de récurrence ($I$ vraie), $(m_0 - m)n = r$. 
			      Donc $(m_0 - m')n = r + n = r'$. La propriété $I$ est donc conservée.
		\end{enumerate}
		En sortie de boucle, la condition $m > 0$ est fausse, donc $m = 0$. L'invariant nous assure alors que $r = (m_0 - 0)n = m_0 n$. L'algorithme calcule bien le produit attendu et est donc correct.
	}
\end{Exercise}


\begin{Exercise}[title = {Retourner un tableau}]\\
	Dans cet exercice, on s'intéresse à un algorithme permettant de \og retourner\fg{} un tableau, c'est-à-dire réorganiser l'ordre de ses éléments de façon que le premier élément devienne le dernier, le deuxième devienne l'avant-dernier et ainsi de suite. Par exemple, le tableau {\tt \{2, 7, 1, 9, 3\}} deviendrait {\tt \{3, 9, 1, 7, 2\}}. On notera bien que le retournement doit s'effectuer \textit{en place}, c'est-à-dire qu'on ne veut pas créer de nouveau tableau mais simplement modifier le tableau existant. En notant $t_0$ le tableau initial et $t_r$ le tableau \og retourné\fg{}, on a donc pour tout $k \in \intN{0}{n-1},\ t_r[k] = t_0[n-1-k]$. On propose pour cela l'algorithme suivant :\\
	\SetKw{KwResult}{Résultat : }
	\begin{algorithm}[H]
		\DontPrintSemicolon
		\caption{Retourner un tableau}
		\Entree{Un tableau $t$ de taille $n$}
		\Sortie{Aucune}
		\Res{Le tableau $t$ est modifié (retourné)}
		\everypar={\footnotesize \textcolor{gray}{\nl}}
		$i \leftarrow 0$\;
		$j \leftarrow n-1$\;
		\Tq{$j-i > 0$}{
			échanger les éléments d'indices $i$ et $j$ dans $t$ \;
			$i \leftarrow i+1$ \;
			$j \leftarrow j-1$ \;
		}
	\end{algorithm}
	\Question{Montrer que cet algorithme termine.}
	\tcor{Dans le cas où le tableau est vide, l'algorithme termine directement sans entrer dans la boucle {\tt while}. Sinon,
		\begin{itemize}
			\item $j-i$ est entier comme différence de deux entiers
			\item $j-i$ est positif avant d'entrer dans la boucle ($n \geqslant 1$) et reste positif par condition d'entrée dans la boucle
			\item $j-i$ décroît strictement puisque $i$ est incrémenté et $j$ décrémenté
		\end{itemize}
	}
	\Question{Montrer que la propriété suivante, notée $I$, est un invariant de l'algorithme : $i+j = n-1$.}
	\tcor{
		\begin{itemize}
		\item Avant d'entrer dans la boucle, $i = 0$ et $j=n-1$, donc $I$ est vraie.
		\item On suppose $I$ vraie au début d'un tour de boucle. Montrons qu'alors $I$ est conservée à la fin de ce passage. En notant $i' = i+1$ et $j' = j-1$ les nouvelles valeurs, on a $i'+j' = i+1+j-1 = i+j$. Or, par hypothèse $i+j = n-1$, donc $i'+j' = n-1$ et $I$ est conservée.
		\end{itemize}
	}
	\Question{Montrer que cet algorithme est correct. On pourra pour cela considérer l'invariant $I'$ : \og \textit{le tableau $t$ contient les valeurs de $t_r$ (le tableau retourné) pour tous les indices $k \in \intN{0}{i-1} \cup \intN{j+1}{n-1}$}\fg{} et utiliser l'invariant $I$ de la question précédente.}
	\tcor{On vérifie que l'invariant proposé est vrai :
		\begin{itemize}
			\item Avant d'entrer dans la boucle, $\intN{0}{i-1} \cup \intN{j+1}{n-1} = \varnothing$ et donc $I'$ est vraie par vacuité.
			\item Conservé lors d'un passage dans la boucle : si $I'$ est vrai en entrant dans la boucle, alors on échange $t[i]$ avec $t[j]$. D'après l'invariant $I$ démontré à la question précédente, $j = n-1-i$. On échange donc $t[i]$ avec $t[n-1-i]$. Comme, par hypothèse, les éléments situés avant $i$ et après $j$ sont déjà correctement échangés ($I'$ supposé vrai), la propriété $I'$ reste vraie pour les indices s'étendant jusqu'à $i$ et $j$.
		\end{itemize}
		En sortie de boucle, on a $j-i \leqslant 0$ et donc $\intN{0}{i-1} \cup \intN{j+1}{n-1} = \intN{0}{n-1}$ ; le tableau $t$ contient bien les valeurs du tableau retourné.
	}
	\Question{On souhaite utiliser cet algorithme afin d'écrire en langage C une fonction {\tt retourner\_str} qui inverse une chaîne de caractères (par exemple, la chaîne {\tt "MP2I"} devient {\tt "I2PM"}). Rappeler la façon dont sont implémentées en C les chaînes de caractères et expliquer pour quelle raison il n'est pas utile de fournir la longueur de la chaîne en paramètre de la fonction {\tt retourner\_str}.}
	\tcor{En C, les chaînes de caractères sont des tableaux de caractères se terminant par le caractère sentinelle {\tt '\textbackslash 0'}. On peut donc obtenir la taille d'une chaîne en recherchant la position de ce caractère ; il n'est donc pas utile de la passer en paramètre.}
	\Question{Écrire une fonction de signature \mintinline{c}{void echange(char s[], int i, int j)} qui échange dans {\tt s} les caractères situés aux indices {\tt i} et {\tt j}.}
	\ifcorrige
		\corpartC{retourner_str.c}{}{}{4}{10}
	\fi
	\Question{Écrire une implémentation de la fonction {\tt retourner\_str} sans utiliser les fonctions de la bibliothèque {\tt <string.h>}.}
	\ifcorrige
		\corpartC{retourner_str.c}{}{}{12}{28}
	\fi
	\Question{On rappelle qu'un palindrome est un mot qui se lit de la même façon de droite à gauche ou de gauche à droite, par exemple \og \textit{radar}\fg{} est un palindrome, mais \og \textit{tata}\fg{} n'en est pas un. Pour tester si un mot est un palindrome, un élève propose la solution suivante :
	\inputpartC{retourner_str.c}{}{}{30}{36}
	Ce programme ne compile pas et produit une erreur à la ligne 4 : \textit{\og error : invalid initializer\fg{}}. Expliquer la cause de cette erreur (on ne demande pas de réécrire la fonction \texttt{palindrome}, mais d'expliquer de façon succincte l'erreur commise à la ligne 4 et la façon d'y remédier).}
	\tcor{
		On ne peut pas affecter directement un tableau en C, on doit travailler élément par élément. Il faudrait donc copier un à un les caractères de {\tt s} dans le tableau {\tt copie}.
	}
	\Question{On suppose corrigée l'erreur commise à la ligne 4. Cette fonction est-elle alors correcte ? Justifier votre réponse.}
	\tcor{
		Non, cette fonction n'est pas correcte. En C, l'expression {\tt s == copie} compare les adresses mémoire des deux tableaux (qui sont distinctes) et non leurs contenus : le test sera donc toujours faux. De plus, il n'existe pas d'opérateur d'égalité globale entre tableaux ; on doit comparer les caractères un à un dans une boucle (ou utiliser {\tt strcmp}).
	}
\end{Exercise}

\begin{Exercise}[title = {Autour de la recherche de minimums}]\\
	On souhaite écrire une fonction de signature \mintinline{c}{int min(int tab[], int taille)} qui prend en entrée un tableau non vide d'entiers {\tt tab} ainsi que sa taille {\tt taille} et renvoie le minimum des éléments de ce tableau. On donne ci-dessous deux réponses d'élèves où les fonctions sont nommées respectivement {\tt min\_eleve1} et {\tt min\_eleve2}.
	\inputpartC{smin.c}{}{}{11}{24}
	\inputpartC{smin.c}{}{}{26}{39}
	\Question{Donner un exemple de tableau à 3 éléments pour lequel {\tt min\_eleve1} renvoie un résultat correct.}
	\tcor{On peut prendre par exemple le tableau {\tt \{5, -2, 1\}} pour lequel la fonction renvoie bien {\tt -2}. En effet, le problème de cette fonction est l'initialisation à 0 du minimum : dès lors que le minimum est négatif ou nul, la valeur renvoyée est correcte.}
	\Question{Proposer un exemple de votre choix montrant que {\tt min\_eleve1} ne répond pas à sa spécification.}
	\tcor{Si le tableau ne contient que des valeurs \textit{strictement positives}, alors, le minimum étant initialisé à 0, le test de la ligne 8 n'est jamais vrai et la valeur renvoyée pour le minimum sera 0. On peut par exemple prendre le tableau {\tt \{2, 7, 5\}} : son minimum est 2, mais la fonction renvoie {\tt 0}.}
	\Question{Donner un exemple de tableau à 3 éléments pour lequel {\tt min\_eleve2} renvoie un résultat correct.}
	\tcor{On peut prendre par exemple le tableau {\tt \{5, -2, -3\}} pour lequel la fonction renvoie bien {\tt -3}. En effet, le problème de cette fonction est de comparer l'élément avec son suivant et non avec le minimum trouvé jusque-là.}
	\Question{Proposer un exemple de votre choix montrant que {\tt min\_eleve2} ne répond pas à sa spécification.}
	\tcor{On peut prendre par exemple le tableau {\tt \{-4, -2, -3\}} pour lequel la fonction renvoie {\tt -3}. En effet, la comparaison entre les deux derniers éléments met à jour {\tt cmin} à {\tt -3}, alors que le minimum est {\tt -4}.}
	\Question{Donner une version de la fonction {\tt min} conforme à sa spécification.}
	\ifcorrige
		\corpartC{smin.c}{}{}{41}{54}
	\fi

	\Question{On veut maintenant obtenir, en plus du minimum, son nombre d'occurrences dans le tableau. Par exemple, pour le tableau {\tt \{1, 0, 7, 2\}}, le minimum est 0 (il apparaît 1 fois) et pour {\tt \{1, 7, -3, 2, -3, 4\}}, le minimum est $-3$ (il apparaît 2 fois). Pour cela, on propose d'écrire une fonction de signature : \mintinline{c}{int min_occ(int tab[], int taille, int *nb)}. Cette fonction renvoie le minimum \textit{et} modifie la valeur pointée par {\tt nb} afin qu'elle contienne le nombre d'apparitions du minimum. Donner une implémentation en langage C de la fonction {\tt min\_occ}.}

	\NRet \smallskip 
	On s'intéresse maintenant à la recherche des deux plus petites valeurs d'un tableau contenant au moins deux éléments (ces deux valeurs pouvant être égales si le minimum apparaît au moins deux fois). 

	\Ret \smallskip

	\Question{Écrire un algorithme (en pseudo-langage) permettant de résoudre ce problème. On écrira une instruction {\tt return} pour renvoyer les deux plus petites valeurs.}
	\tcor{
		\begin{algorithm}[H]
		\DontPrintSemicolon
		\caption{Deux minimums}
		\Entree{Un tableau $t$ de $n$ entiers avec $n \geqslant 2$.}
		\Sortie{Les deux plus petites valeurs de $t$}
		\everypar={\footnotesize \textcolor{gray}{\nl}}
		\Si {$t[0] < t[1]$}
		 {$m1 \leftarrow t[0]$\;
		  $m2 \leftarrow t[1]$\;}
		\Sinon
		{$m1 \leftarrow t[1]$\;
		  $m2 \leftarrow t[0]$\;}
		\Pour{$i \leftarrow 2$ \KwA $n-1$}{
			\Si {$t[i] < m1$}
			{
				$m2 \leftarrow m1$\;
				$m1 \leftarrow t[i]$\;
			}
			\SinonSi {$t[i] < m2$}
			{$m2 \leftarrow t[i]$ \;}
		}
		\Return $m1, m2$
	\end{algorithm}
	}
	\Question{En C, on ne peut pas directement renvoyer deux entiers, aussi on propose de passer à la fonction deux pointeurs vers des entiers dont le contenu sera mis à jour afin de contenir après appel les deux plus petites valeurs. Écrire cette fonction.}
	\ifcorrige
		\corpartC{smin.c}{}{}{106}{125}
	\fi
	\Question{Proposer un invariant de boucle qui permettrait de prouver la correction de cet algorithme. On ne demande pas de faire une preuve de correction, mais simplement d'énoncer un invariant préservé durant le déroulement de l'algorithme.}
	\tcor{L'invariant à prouver est qu'au début de chaque itération d'indice $i$ (pour $2 \leqslant i \leqslant n$), $m1$ et $m2$ contiennent les deux plus petites valeurs du sous-tableau $t[0 \dots i-1]$ avec $m1 \leqslant m2$. En fin d'algorithme, puisque $i = n$, ces deux variables contiennent bien les deux plus petites valeurs de l'ensemble du tableau, ce qui garantit la correction.}
\end{Exercise}
