	\Question{Écrire une fonction de signature \mintinline{c}{int identiques(char s[], char t[])} qui renvoie le nombre de fois où les chaînes passées en argument ont le même caractère au même emplacement. Par exemple, 
	\begin{itemize}
		\item \mintinline{c}{identiques("Minimum", "Maximum")} renvoie 5, car les deux chaines coïncident sur {\tt M} puis sur {\tt imum}.
		\item \mintinline{c}{identiques("Maths","Informatique")} renvoie 0.
		\item \mintinline{c}{identiques("MP2I","PCSI")} renvoie 1.
	\end{itemize}}
	\reponse{8}{7}
	\ifcorrige
	\inputpartC{identiques.c}{}{}{14}{26}
	\fi
