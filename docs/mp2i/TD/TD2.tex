\begin{Exercise}[title = {Étude d'un algorithme}]\\
	On considère l'algorithme suivant : \\
	\SetAlFnt{\small}
	\setlength{\algomargin}{8pt}
	\begin{algorithm}[H]
		\DontPrintSemicolon
		\caption{Quotient dans la division euclidienne}
		\Entree{$a \in \N, b \in \N^*$}
		\Sortie{$q \in \N$, quotient de $a$ par $b$}
		\everypar={\footnotesize \textcolor{gray}{\nl}}
		$q \leftarrow 0$\;
		$r \leftarrow a$\;
		\Tq{$r \geqslant 0$}{
		$q \leftarrow q+1$ \;
		$r \leftarrow r-b$ \;
		}
		\Return $q - 1$
	  \end{algorithm}

\Question{Proposer une implémentation de cet algorithme en langage C, sous la forme d'une fonction dont on précisera soigneusement la spécification.}
\Question{Étudier la terminaison de cet algorithme.}
\Question{Étudier la correction de cet algorithme.}
\end{Exercise}

\begin{Exercise}[title = {Algorithme d'Euclide}]\\
	On considère l'algorithme d'Euclide par soustractions pour calculer le plus grand commun diviseur ($\mathrm{pgcd}$) de deux entiers strictement positifs :\\
	\SetAlFnt{\small}
	\setlength{\algomargin}{8pt}
	\begin{algorithm}[H]
		\DontPrintSemicolon
		\caption{PGCD par soustractions successives}
		\Entree{$a \in \N^*, b \in \N^*$}
		\Sortie{$\mathrm{pgcd}(a, b)$}
		\everypar={\footnotesize \textcolor{gray}{\nl}}
		\Tq{$a \neq b$}{
			\Si{$a > b$}{
				$a \leftarrow a - b$ \;
			}
			\Sinon{
				$b \leftarrow b - a$ \;
			}
		}
		\Return $a$
	\end{algorithm}
	\Question{Donner une implémentation en C de cet algorithme.}
	\Question{Montrer que $a + b$ est un variant de boucle et en déduire la terminaison de l'algorithme.}
	\Question{Montrer que la propriété \og $\mathrm{pgcd}(a, b) = \mathrm{pgcd}(a_0, b_0)$ \fg{} est un invariant de boucle, où $a_0$ et $b_0$ désignent les valeurs initiales de $a$ et $b$. En déduire la correction totale de l'algorithme.}
\end{Exercise}

\begin{Exercise}[title = {Recherche dichotomique}]\\
	On considère un tableau $t$ de $n$ entiers trié par ordre strictement croissant et une valeur cible $x$. On souhaite déterminer si $x$ appartient à $t$ :\\
	\SetAlFnt{\small}
	\setlength{\algomargin}{8pt}
	\begin{algorithm}[H]
		\DontPrintSemicolon
		\caption{Recherche dichotomique}
		\Entree{$t$ tableau d'entiers de taille $n$ trié par ordre croissant, $x$ entier}
		\Sortie{Indice $m$ tel que $t[m] = x$, ou $-1$ si $x$ n'appartient pas à $t$}
		\everypar={\footnotesize \textcolor{gray}{\nl}}
		$g \leftarrow 0$ \;
		$d \leftarrow n - 1$ \;
		\Tq{$g \leqslant d$}{
			$m \leftarrow g + \lfloor (d - g) / 2 \rfloor$ \;
			\Si{$t[m] = x$}{\Return $m$}
			\SinonSi{$t[m] < x$}{$g \leftarrow m + 1$}
			\Sinon{$d \leftarrow m - 1$}
		}
		\Return $-1$
	\end{algorithm}
	\Question{Proposer un jeu de tests complet en utilisant la méthode du partitionnement des données d'entrée et en testant les cas limites.}
	\Question{Montrer que la quantité $d - g + 1$ permet de prouver la terminaison de la boucle.}
	\Question{Prouver que la propriété \og si $x$ est présent dans le tableau $t$, alors son indice est dans $\llbracket g, d \rrbracket$ \fg{} est un invariant de boucle. En déduire la correction totale de l'algorithme.}
\end{Exercise}

\begin{Exercise}[title = {Exponentiation rapide}]\\
	On considère l'algorithme suivant calculant $x^n$ pour $x \in \R$ et $n \in \N$ :\\
	\SetAlFnt{\small}
	\setlength{\algomargin}{8pt}
	\begin{algorithm}[H]
		\DontPrintSemicolon
		\caption{Exponentiation rapide itérative}
		\Entree{$x \in \R, n \in \N$}
		\Sortie{$x^n$}
		\everypar={\footnotesize \textcolor{gray}{\nl}}
		$r \leftarrow 1.0$ \;
		$p \leftarrow x$ \;
		$k \leftarrow n$ \;
		\Tq{$k > 0$}{
			\Si{$k \bmod 2 = 1$}{
				$r \leftarrow r \times p$ \;
			}
			$p \leftarrow p \times p$ \;
			$k \leftarrow \lfloor k / 2 \rfloor$ \;
		}
		\Return $r$
	\end{algorithm}
	\Question{Faire tourner cet algorithme à la main pour $x = 3.0$ et $n = 11$ en précisant les valeurs prises par $r$, $p$ et $k$ à chaque étape.}
	\Question{Proposer une implémentation en langage C sous la forme d'une fonction \mintinline{c}{double puissance(double x, int n)} accompagnée de ses préconditions avec \mintinline{c}{assert}.}
	\Question{Prouver la terminaison de cet algorithme.}
	\Question{Prouver que la propriété $I$ : \og{} $r \times p^k = x^n$ \fg{} est un invariant de boucle. Conclure sur la correction de l'algorithme.}
\end{Exercise}
